cmake_minimum_required(VERSION 3.23)
project(svf_cegar_lab LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SVF REQUIRED CONFIG)
# Some in-tree SVF builds export an unused include/SVF directory before install.
get_target_property(_svf_includes SVF::SvfFlags INTERFACE_INCLUDE_DIRECTORIES)
set(_existing_includes "")
foreach(_dir IN LISTS _svf_includes)
  if(EXISTS "${_dir}")
    list(APPEND _existing_includes "${_dir}")
  endif()
endforeach()
set_property(TARGET SVF::SvfFlags PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${_existing_includes}")
add_executable(svf-cegar src/checker.cpp)
target_link_libraries(svf-cegar PRIVATE SVF::SvfLLVM)
target_compile_options(svf-cegar PRIVATE -Wno-error)
add_executable(smt-replay src/smt-replay.cpp)
target_include_directories(smt-replay PRIVATE "${Z3_INCLUDE_DIR}")
target_link_libraries(smt-replay PRIVATE "${Z3_LIBRARY_DIR}")
